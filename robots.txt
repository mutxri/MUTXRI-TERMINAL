# Only https://mutxriterminal.com/ should appear in search results.
#
# Everything else - the terminal, its panels and the docs - carries
# <meta name="robots" content="noindex, nofollow"> instead of being blocked
# here. That is deliberate: Disallow stops a crawler FETCHING a page, so
# Googlebot never reads the noindex tag and any URL it discovered through a
# link stays in the index indefinitely. That is exactly what happened - the
# terminal and the heatmap panel were listed in search while blocked here.
#
# Letting crawlers fetch those pages is what allows them to see noindex and
# drop them. Do not re-add Disallow lines for them.

User-agent: *
Allow: /

Sitemap: https://mutxriterminal.com/sitemap.xml
